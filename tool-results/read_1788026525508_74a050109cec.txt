   152→function JobCardDetail({ jc, money, onClose, onOpenInvoice }: { jc: any; money: (n: number) => string; onClose: () => void; onOpenInvoice?: (id: string) => void }) {
   153→  const { t, lang } = useT();
   154→  const { invalidate, toastSuccess, toastError } = useApiMutation();
   155→  const [notes, setNotes] = useState(jc.diagnosis || "");
   156→
   157→  const nextStatus = (s: string) => {
   158→    const idx = STATUSES.indexOf(s);
   159→    return STATUSES[idx + 1] && STATUSES[idx + 1] !== "cancelled" ? STATUSES[idx + 1] : null;
   160→  };
   161→  const next = nextStatus(jc.status);
   162→
   163→  const advance = async (status: string) => {
   164→    try {
   165→      await fetch(`/api/job-cards/${jc.id}`, { method: "PATCH", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ status, diagnosis: notes }) });
   166→      invalidate(["/api/job-cards", "/api/dashboard", "/api/parts"]);
   167→      toastSuccess();
   168→      onClose();
   169→    } catch {
   170→      toastError("Error");
   171→    }
   172→  };
   173→
   174→  return (
   175→    <div className="space-y-4">
   176→      <div className="grid grid-cols-2 gap-2 sm:grid-cols-4 text-xs">
   177→        <Info label={t("customer")} value={jc.customer?.name} />
   178→        <Info label={t("vehicle")} value={`${jc.vehicle?.plateNumber} · ${jc.vehicle?.make} ${jc.vehicle?.model}`} />
   179→        <Info label={t("mileage")} value={`${formatNumber(jc.mileage, lang)} km`} />
   180→        <Info label={t("priority")} value={t("status_" + jc.priority)} />
   181→        <Info label={t("technician")} value={jc.technician?.name || "—"} />
   182→        <Info label={t("estimatedCompletion")} value={jc.estimatedCompletion ? formatDate(jc.estimatedCompletion, lang) : "—"} />
   183→        <Info label={t("date")} value={formatDate(jc.createdAt, lang)} />
   184→        <Info label={t("grandTotal")} value={money(jc.grandTotal)} />
   185→      </div>
   186→
   187→      <div className="rounded-lg border p-3">
   188→        <p className="text-xs font-medium text-muted-foreground mb-1">{t("complaint")}</p>
   189→        <p className="text-sm">{jc.complaint}</p>
   190→      </div>
   191→
   192→      <div>
   193→        <p className="text-xs font-medium text-muted-foreground mb-1">{t("diagnosis")}</p>
   194→        <Textarea value={notes} onChange={(e) => setNotes(e.target.value)} placeholder={t("diagnosis")} rows={2} />

   196→
   197→      {/* Services & parts */}
   198→      <div className="grid gap-3 md:grid-cols-2">
   199→        <div>
   200→          <h4 className="mb-2 text-sm font-semibold">{t("labor")} ({jc.services?.length ?? 0})</h4>

   202→            <Table>
   203→              <TableHeader><TableRow><TableHead className="text-xs">{t("services")}</TableHead><TableHead className="text-xs text-end">{t("total")}</TableHead></TableRow></TableHeader>
   204→              <TableBody>
   205→                {jc.services?.map((s: any) => (
   206→                  <TableRow key={s.id}>
   207→                    <TableCell className="text-xs"><span className="font-medium">{s.name}</span><span className="block text-muted-foreground">{s.hours}h</span></TableCell>
   208→                    <TableCell className="text-xs text-end tnum font-medium">{money(s.total)}</TableCell>
   209→                  </TableRow>
   210→                ))}
   211→              </TableBody>
   212→            </Table>
   213→          </div>
   214→        </div>
   215→        <div>
   216→          <h4 className="mb-2 text-sm font-semibold">{t("parts")} ({jc.parts?.length ?? 0})</h4>
   217→          <div className="overflow-x-auto rounded-lg border">
   218→            <Table>
   219→              <TableHeader><TableRow><TableHead className="text-xs">{t("name")}</TableHead><TableHead className="text-xs text-end">{t("quantity")}</TableHead><TableHead className="text-xs text-end">{t("total")}</TableHead></TableRow></TableHeader>

   222→                  <TableRow key={p.id}>
   223→                    <TableCell className="text-xs"><span className="font-medium">{p.part?.name}</span><StatusBadge status={p.status} label={p.status} /></TableCell>
   224→                    <TableCell className="text-xs text-end tnum">{p.quantity}</TableCell>
   225→                    <TableCell className="text-xs text-end tnum font-medium">{money(p.total)}</TableCell>
   226→                  </TableRow>
   227→                ))}
   228→              </TableBody>
   229→            </Table>
   230→          </div>
   231→        </div>
   232→      </div>
   233→
   234→      {/* Totals */}
   235→      <div className="ms-auto w-full max-w-xs space-y-1 text-sm">
   236→        <Row label={t("laborTotal")} value={money(jc.laborTotal)} />

   252→        </Select>
   253→        {next && (
   254→          <Button size="sm" variant="default" onClick={() => advance(next)}>
   255→            <ArrowRight className="h-3.5 w-3.5 me-1" />{t("status_" + next)}
   256→          </Button>
   257→        )}
   258→        <Button size="sm" variant="outline" className="ms-auto no-print" onClick={() => window.print()}><Printer className="h-3.5 w-3.5 me-1" />{t("print")}</Button>
   259→      </div>
   260→
   261→      {/* Auto-invoice banner — the invoice is created automatically when the job card is created */}
   262→      {jc.invoice ? (
   263→        <div className="flex flex-wrap items-center gap-2 rounded-lg border border-emerald-300/60 bg-emerald-50 p-3 dark:border-emerald-900/40 dark:bg-emerald-950/30">
   264→          <Receipt className="h-5 w-5 shrink-0 text-emerald-600" />
   265→          <div className="min-w-0 flex-1">
   266→            <p className="text-sm font-semibold text-emerald-700 dark:text-emerald-300">{t("autoInvoiceCreated")}</p>
   267→            <p className="text-xs text-emerald-600/80 dark:text-emerald-400/80">{t("autoInvoiceHint")}</p>
   268→          </div>
   269→          <span className="rounded-full bg-emerald-100 px-2.5 py-1 font-mono text-xs font-bold text-emerald-800 dark:bg-emerald-950 dark:text-emerald-300 tnum">{jc.invoice.code}</span>

   271→            <Button size="sm" variant="outline" className="h-7 text-xs border-emerald-300 text-emerald-700 hover:bg-emerald-100 dark:border-emerald-800 dark:text-emerald-300" onClick={() => onOpenInvoice(jc.invoice.id)}>
   272→              <FileText className="h-3.5 w-3.5 me-1" />{t("viewInvoice")}
   273→            </Button>
   274→          )}
   275→        </div>
   276→      ) : null}
   277→    </div>
   278→  );
   279→}

   295→function JobCardCreateDialog({ open, onOpenChange }: { open: boolean; onOpenChange: (o: boolean) => void }) {
   296→  const { t, lang } = useT();
   297→  const { data: cust } = useApi<any>("/api/customers");
   298→  const { data: svc } = useApi<any>("/api/services");
   299→  const { data: parts } = useApi<any>("/api/parts");
   300→  const { data: tenants } = useApi<any>("/api/settings");
   301→  const taxPercent = tenants?.taxPercent ?? 0;
   302→  const { invalidate, toastSuccess, toastError } = useApiMutation();
   303→  const [saving, setSaving] = useState(false);
   304→
   305→  const [customerId, setCustomerId] = useState("");
   306→  const [vehicleId, setVehicleId] = useState("");
   307→  const [complaint, setComplaint] = useState("");
   308→  const [diagnosis, setDiagnosis] = useState("");
   309→  const [technicianId, setTechnicianId] = useState("");
   310→  const [priority, setPriority] = useState("normal");
   311→  const [estimatedCompletion, setEstimatedCompletion] = useState("");
   312→  const [selectedServices, setSelectedServices] = useState<string[]>([]);
   313→  const [customServices, setCustomServices] = useState<Array<{ id: string; name: string; price: number }>>([]);
   314→  const [selectedParts, setSelectedParts] = useState<Record<string, number>>({});
   315→
   316→  const customers = cust?.items || [];
   317→  const services = svc?.items || [];
   318→  const allParts = parts?.items || [];
   319→  const selectedCustomer = customers.find((c: any) => c.id === customerId) || null;
   320→  const vehicles = selectedCustomer?.vehicles || [];
   321→
   322→  const calc = useMemo(() => {
   323→    const svcs = services.filter((s: any) => selectedServices.includes(s.id));
   324→    const catalogLabor = svcs.reduce((sum: number, s: any) => sum + s.price, 0);
   325→    const customLabor = customServices.reduce((sum: number, c) => sum + (Number(c.price) || 0), 0);
   326→    const laborTotal = catalogLabor + customLabor;
   327→    const partItems = Object.entries(selectedParts).map(([pid, qty]) => {
   328→      const p = allParts.find((x: any) => x.id === pid);
   329→      return p ? { ...p, qty } : null;
   330→    }).filter(Boolean);
   331→    const partsTotal = partItems.reduce((sum: number, p: any) => sum + p.sellingPrice * p.qty, 0);
   332→    const subtotal = laborTotal + partsTotal;
   333→    const tax = Math.round((subtotal * taxPercent) / 100 * 1000) / 1000;
   334→    const grand = Math.round((subtotal + tax) * 1000) / 1000;
   335→    return { svcs, partItems, laborTotal, partsTotal, tax, grand, customLabor };
   336→  }, [selectedServices, selectedParts, services, allParts, taxPercent, customServices]);
   337→
   338→  const reset = () => {
   339→    setCustomerId(""); setVehicleId(""); setComplaint(""); setDiagnosis(""); setTechnicianId(""); setPriority("normal"); setEstimatedCompletion(""); setSelectedServices([]); setSelectedParts({}); setCustomServices([]);
   340→  };
   341→
   342→  const submit = async () => {
   343→    if (!customerId || !vehicleId || !complaint) return toastError(t("required"));
   344→    setSaving(true);
   345→    try {
   346→      const servicesPayload = [
   347→        ...calc.svcs.map((s: any) => ({ serviceId: s.id, name: s.name, hours: s.durationHours, laborPrice: s.price, discount: 0, total: s.price })),
   348→        ...customServices.map((c) => ({ serviceId: null, name: c.name, hours: 0, laborPrice: Number(c.price) || 0, discount: 0, total: Number(c.price) || 0 })),
   349→      ];
   350→      const jobParts = calc.partItems.map((p: any) => ({ partId: p.id, quantity: p.qty, unitPrice: p.sellingPrice, discount: 0, total: p.sellingPrice * p.qty }));
   351→      await fetch("/api/job-cards", {
   352→        method: "POST",
   353→        headers: { "Content-Type": "application/json" },
   354→        body: JSON.stringify({ customerId, vehicleId, complaint, diagnosis, technicianId, priority, estimatedCompletion, services: servicesPayload, parts: jobParts, status: "draft" }),
   355→      });
   356→      invalidate(["/api/job-cards", "/api/parts", "/api/dashboard"]);
   357→      toastSuccess();
   358→      reset();
   359→      onOpenChange(false);
   360→    } catch {
   361→      toastError("Error");
   362→    } finally {
   363→      setSaving(false);
   364→    }
   365→  };
   366→
   367→  return (
   368→    <Dialog open={open} onOpenChange={onOpenChange}>
   369→      <DialogContent className="max-w-3xl max-h-[92vh] overflow-y-auto scroll-thin">
   370→        <DialogHeader>
   371→          <DialogTitle className="flex items-center gap-2"><Wrench className="h-5 w-5" />{t("addNew")} · {t("jobCards")}</DialogTitle>
   372→          <DialogDescription>{t("complaint")} → {t("inspection")} → {t("estimateFor")} → {t("jobCards")}</DialogDescription>
   373→        </DialogHeader>
   374→
   375→        <div className="space-y-4">
   376→          <div className="grid gap-3 sm:grid-cols-2">
   377→            <Field label={`${t("customer")} *`}>
   378→              <CustomerPicker
   379→                customers={customers}
   380→                selected={selectedCustomer}
   381→                onSelect={(c) => { setCustomerId(c.id); setVehicleId(""); }}
   382→                onClear={() => { setCustomerId(""); setVehicleId(""); }}
   383→              />
   384→            </Field>
   385→            <Field label={`${t("vehicle")} *`}>
   386→              <Select value={vehicleId} onValueChange={setVehicleId} disabled={!customerId}>
   387→                <SelectTrigger><SelectValue placeholder={t("vehicle")} /></SelectTrigger>
   388→                <SelectContent>
   389→                  {vehicles.map((v: any) => <SelectItem key={v.id} value={v.id}>{v.plateNumber} · {v.make} {v.model}</SelectItem>)}
   390→                </SelectContent>
   391→              </Select>
   392→            </Field>
   393→            <Field label={t("complaint")}>
   394→              <Input value={complaint} onChange={(e) => setComplaint(e.target.value)} placeholder={t("complaint")} />
   395→            </Field>
   396→            <Field label={t("diagnosis")}>
   397→              <Input value={diagnosis} onChange={(e) => setDiagnosis(e.target.value)} placeholder={t("diagnosis")} />
   398→            </Field>
   399→            <Field label={t("priority")}>
   400→              <Select value={priority} onValueChange={setPriority}>
   401→                <SelectTrigger><SelectValue /></SelectTrigger>
   402→                <SelectContent>

   409→            </Field>

   427→          <PartPicker

   429→            selected={selectedParts}
   430→            onToggle={(id) => setSelectedParts(selectedParts[id] ? ({ ...selectedParts, [id]: undefined }) : { ...selectedParts, [id]: 1 })}
   431→            onQty={(id, qty) => setSelectedParts({ ...selectedParts, [id]: Math.max(1, qty) })}
   432→            onClear={() => setSelectedParts({})}
   433→          />
   434→
   435→          {/* Totals */}
   436→          <div className="ms-auto w-full max-w-xs space-y-1 rounded-lg bg-muted/50 p-3 text-sm">
   437→            <Row label={t("laborTotal")} value={formatMoney(calc.laborTotal, "OMR", lang)} />
   438→            <Row label={t("partsTotal")} value={formatMoney(calc.partsTotal, "OMR", lang)} />
   439→            <Row label={t("tax")} value={formatMoney(calc.tax, "OMR", lang)} />
   440→            <div className="flex justify-between border-t pt-1 text-base font-bold"><span>{t("grandTotal")}</span><span className="tnum">{formatMoney(calc.grand, "OMR", lang)}</span></div>
   441→          </div>
   442→
   443→          <div className="flex justify-end gap-2 pt-1">
   444→            <Button variant="outline" onClick={() => onOpenChange(false)}>{t("cancel")}</Button>
   445→            <Button onClick={submit} disabled={saving}>{t("create")}</Button>
   446→          </div>
   447→        </div>
   448→      </DialogContent>
   449→    </Dialog>
   450→  );

   453→// ─── Customer Picker (search by name / phone / plate) ──────────
   454→function CustomerPicker({ customers, selected, onSelect, onClear }: { customers: any[]; selected: any; onSelect: (c: any) => void; onClear: () => void }) {
   455→  const { t, lang } = useT();

   504→              <button
   505→                key={c.id}
   506→                type="button"
   507→                onMouseDown={(e) => e.preventDefault()}
   508→                onClick={() => { onSelect(c); setQuery(""); setOpen(false); }}
   509→                className="flex w-full items-center gap-2.5 rounded px-2 py-2 text-start hover:bg-accent"
   510→              >
   511→                <div className="flex h-8 w-8 shrink-0 items-center justify-center rounded-full bg-primary/10 text-xs font-bold text-primary">
   512→                  {c.name.split(" ").map((n: string) => n[0]).join("").slice(0, 2)}
   513→                </div>
   514→                <div className="min-w-0 flex-1">
   515→                  <p className="truncate text-sm font-medium">{c.name}</p>
   516→                  <p className="truncate text-xs text-muted-foreground">{c.mobile}</p>
   517→                </div>
   518→                {(c.vehicles?.length ?? 0) > 0 && (
   519→                  <span className="shrink-0 rounded-full bg-muted px-1.5 py-0.5 text-[10px] font-medium text-muted-foreground tnum">
   520→                    {c.vehicles.map((v: any) => v.plateNumber).join(" · ")}
   521→                  </span>
   522→                )}
   523→              </button>
   524→            ))
   525→          )}
   526→        </div>
   527→      )}
   528→    </div>
   529→  );

   664→// ─── Custom Services editor (manual name + price) ──────────────
   665→function CustomServicesEditor({ services, onChange }: {
   666→  services: Array<{ id: string; name: string; price: number }>;
   667→  onChange: (services: Array<{ id: string; name: string; price: number }>) => void;
   668→}) {
   669→  const { t, lang } = useT();
   670→  const [name, setName] = useState("");
   671→  const [price, setPrice] = useState("");
   672→
   673→  const add = () => {
   674→    if (!name.trim()) return;
   675→    const newService = { id: `custom-${Date.now()}`, name: name.trim(), price: Number(price) || 0 };
   676→    onChange([...services, newService]);
   677→    setName("");
   678→    setPrice("");
   679→  };
   680→
   681→  const remove = (id: string) => onChange(services.filter((s) => s.id !== id));
   682→
   683→  const update = (id: string, field: "name" | "price", value: string) => {
   684→    onChange(services.map((s) => s.id === id ? { ...s, [field]: field === "price" ? (Number(value) || 0) : value } : s));
   685→  };
   686→
   687→  return (
   688→    <div>
   689→      <div className="mb-2 flex items-center justify-between">
   690→        <p className="text-sm font-semibold">{t("otherServices")} <span className="text-muted-foreground">({services.length})</span></p>
   691→        {services.length > 0 && (
   692→          <button type="button" onClick={() => onChange([])} className="text-xs text-muted-foreground hover:text-foreground">{t("remove")} ({services.length})</button>
   693→        )}
   694→      </div>
   695→      <p className="mb-2 text-[10px] text-muted-foreground">{t("otherServicesHint")}</p>
   696→
   697→      {/* Add new custom service row */}
   698→      <div className="flex items-center gap-2 rounded-lg border bg-muted/20 p-2">
   699→        <Input
   700→          value={name}
   701→          onChange={(e) => setName(e.target.value)}
   702→          placeholder={t("customServiceNamePlaceholder")}

   706→        <div className="relative w-28">
   707→          <Input
   708→            type="number"
   709→            step="0.001"
   710→            value={price}
   711→            onChange={(e) => setPrice(e.target.value)}
   712→            placeholder="0.000"
   713→            className="h-8 text-end text-xs tnum"

   716→          <span className="pointer-events-none absolute end-2 top-1/2 -translate-y-1/2 text-[10px] text-muted-foreground">OMR</span>

   719→          <Plus className="h-3.5 w-3.5 me-1" />{t("add")}
   720→        </Button>
   721→      </div>

   724→      {services.length === 0 ? (
   725→        <p className="mt-2 text-center text-xs text-muted-foreground py-3">{t("noCustomServices")}</p>
   726→      ) : (
   727→        <div className="mt-2 space-y-1.5">

   729→            <div key={s.id} className="flex items-center gap-2 rounded-lg border bg-card p-2">

   733→                className="h-7 flex-1 text-xs"
   734→              />
   735→              <div className="relative w-28">
   736→                <Input
   737→                  type="number"
   738→                  step="0.001"
